  - platform: template

  - platform: template
    name: "Cell Voltage Delta"
    id: cell_voltage_delta
    unit_of_measurement: "V"
    accuracy_decimals: 3
    lambda: |-
      float max_v = max({
        id(cell_01).state, id(cell_02).state, id(cell_03).state, 
        id(cell_04).state, id(cell_05).state, id(cell_06).state, 
        id(cell_07).state, id(cell_08).state, id(cell_09).state, 
        id(cell_10).state, id(cell_11).state, id(cell_12).state, 
        id(cell_13).state, id(cell_14).state, id(cell_15).state
      });
      float min_v = min({
        id(cell_01).state, id(cell_02).state, id(cell_03).state, 
        id(cell_04).state, id(cell_05).state, id(cell_06).state, 
        id(cell_07).state, id(cell_08).state, id(cell_09).state, 
        id(cell_10).state, id(cell_11).state, id(cell_12).state, 
        id(cell_13).state, id(cell_14).state, id(cell_15).state
      });
      return max_v - min_v;

  # Hidden raw sensor to evaluate the battery state
  - platform: modbus_controller
    modbus_controller_id: esm_bms
    name: "Battery State Raw"
    internal: true
    id: battery_state_raw
    address: 0x000A
    register_type: holding
    value_type: U_WORD
    on_value:
      then:
        - lambda: |-
            if (x == 0) id(battery_state_text).publish_state("Charging");
            else if (x == 1) id(battery_state_text).publish_state("Discharging");
            else if (x == 2) id(battery_state_text).publish_state("Full");
            else if (x == 3) id(battery_state_text).publish_state("Free");
            else id(battery_state_text).publish_state("Unknown");

# Interval automation to check thresholds and compile status into a single text entity
interval:
  - interval: 10s
    then:
      - lambda: |-
          std::string alerts = "";

          // Check High Cell Voltage (Threshold: > 3.65V)
          if (!std::isnan(id(max_cell_voltage).state) && id(max_cell_voltage).state > 3.65) {
            alerts += "[High Cell V] ";
          }

          // Check Low Cell Voltage (Threshold: < 2.80V)
          if (!std::isnan(id(min_cell_voltage).state) && id(min_cell_voltage).state < 2.80) {
            alerts += "[Low Cell V] ";
          }

          // Check Cell Imbalance / Delta (Threshold: > 0.100V / 100mV)
          if (!std::isnan(id(cell_voltage_delta).state) && id(cell_voltage_delta).state > 0.100) {
            alerts += "[Cell Imbalance] ";
          }

          // Check High Temperature (Threshold: > 55°C)
          if (!std::isnan(id(highest_cell_temp).state) && id(highest_cell_temp).state > 55.0) {
            alerts += "[High Temp] ";
          }

          // Check Low Temperature (Threshold: < 0°C)
          if (!std::isnan(id(lowest_cell_temp).state) && id(lowest_cell_temp).state < 0.0) {
            alerts += "[Low Temp] ";
          }

          // Check Low State of Charge (Threshold: < 15%)
          if (!std::isnan(id(bms_soc).state) && id(bms_soc).state < 15.0) {
            alerts += "[Low SOC] ";
          }

          // Publish final state string
          if (alerts.empty()) {
            id(battery_active_alarms).publish_state("Normal");
          } else {
            id(battery_active_alarms).publish_state(alerts);
          }
