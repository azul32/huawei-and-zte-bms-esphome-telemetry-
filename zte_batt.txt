    id: max_cell_voltage
    unit_of_measurement: "V"
    accuracy_decimals: 3
    lambda: |-
      float cells[] = {
        id(cell_1).state, id(cell_2).state, id(cell_3).state,
        id(cell_4).state, id(cell_5).state, id(cell_6).state,
        id(cell_7).state, id(cell_8).state, id(cell_9).state,
        id(cell_10).state, id(cell_11).state, id(cell_12).state,
        id(cell_13).state, id(cell_14).state, id(cell_15).state
      };
      
      float max_v = -1.0;
      for (int i = 0; i < 15; i++) {
        // Only process cells with valid values (not NAN and greater than 0)
        if (!std::isnan(cells[i]) && cells[i] > 0) {
          if (max_v == -1.0 || cells[i] > max_v) {
            max_v = cells[i];
          }
        }
      }
      
      if (max_v == -1.0) {
        return NAN;
      }
      return max_v;

# F. TEXT SENSOR FOR BMS STATUS CODE MAPPING
# ==========================================
text_sensor:
  - platform: template
    name: "BMS Operational Status"
    id: bms_status_text
    update_interval: 5s
    lambda: |-
      float status = id(bms_status_code).state;
      
      if (std::isnan(status)) {
        return std::string("Unknown");
      }
      
      int s = (int)status;
      
      if (s == 0) {
        return std::string("Standby");
      }
      if (s == 2) {
        return std::string("Charging");
      }
      if (s == 1) {
        return std::string("Discharging");
      }
      if (s == 3) {
        return std::string("Fully Charged");
      }
      if (s == 4) {
        return std::string("Fault / Protection");
      }
      
      // Absolute fallback return in the last line to prevent compiler errors
      return std::string("Code: ") + std::to_string(s);
